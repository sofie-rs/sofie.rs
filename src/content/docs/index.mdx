import { Card, CardGrid, Aside } from '@astrojs/starlight/components';

<Aside type="caution" title="Proof of concept">
APIs change without notice. Do not build on sofie for production yet.
</Aside>

<CardGrid>
	<Card title="Location before annotation" icon="document">
		A file’s place says what it is. `sofie gen` reads the app once and writes its wiring; there
		is no container, reflection or service locator at runtime.
	</Card>
	<Card title="Secure by default" icon="approve-check">
		CSRF protection, sealed sessions, an allowlist for every outgoing request and masked
		telemetry are on without configuration.
	</Card>
	<Card title="One database" icon="setting">
		Postgres or SQLite holds the events, jobs, schedules and caches. Redis and Meilisearch are
		optional.
	</Card>
	<Card title="Checked at build time" icon="warning">
		`sofie check` holds every app to the structure, so a module reaching into another one fails
		`cargo build`.
	</Card>
</CardGrid>
